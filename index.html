📋 SQLite Database Schema Architecture

TABLE products: (id TEXT PK, name TEXT, sku TEXT UNIQUE, category TEXT, price REAL, cost REAL, stock INTEGER, min_stock_alert INTEGER, unit TEXT, description TEXT)

TABLE orders: (id TEXT PK, bill_number TEXT UNIQUE, customer_name TEXT, customer_phone TEXT, notes TEXT, subtotal REAL, discount_amount REAL, tax_amount REAL, grand_total REAL, payment_method TEXT, order_date TEXT, status TEXT)

TABLE order_items: (id INTEGER PK AUTOINCREMENT, order_id TEXT FK, product_id TEXT FK, name TEXT, sku TEXT, unit TEXT, price REAL, quantity INTEGER, total REAL)

TABLE supplier_purchase_orders: (id TEXT PK, po_number TEXT UNIQUE, supplier_name TEXT, product_id TEXT, product_name TEXT, quantity INTEGER, cost_per_unit REAL, total_cost REAL, status TEXT, created_at TEXT)

TABLE stock_logs: (id TEXT PK, timestamp TEXT, action_type TEXT, product_id TEXT, product_name TEXT, delta INTEGER, new_stock INTEGER, reference_id TEXT, details TEXT)
